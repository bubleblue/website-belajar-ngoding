Kamus Sintaks Pseudocode & Golang

1. Struktur Utama

program <nama_program>
kamus
    // deklarasi variabel, konstanta, fungsi/prosedur
algoritma
    // deretan instruksi utama
endprogram

2. Deklarasi & Tipe Data (di dalam kamus)

// Variabel dasar:
<nama_variabel> : integer
<nama_variabel> : real
<nama_variabel> : string
<nama_variabel> : char
<nama_variabel> : boolean

// Konstanta & Larik (Array):
const <nama_konstanta> = <nilai>
<nama_array> : array [<awal>..<akhir>] of <tipe_data>
// Contoh: deret : array [1..5] of integer

3. Input, Output & Penugasan

// Masukan (Input):
input(<nama_variabel>)
input(a, b)
input(deret[1])

// Keluaran (Output):
output(<nilai_atau_variabel>)
output("Hasil: ", luas)

// Penugasan (Assignment):
<nama_variabel> <- <nilai>
deret[1] <- 25

4. Percabangan (Branching)

// Bentuk 1 (IF):
if <kondisi> then ... endif

// Bentuk 2 (IF - ELSE):
if <kondisi> then ... else ... endif

// Bentuk 3 (IF - ELSE IF - ELSE):
if <kondisi> then
    ...
else if <kondisi> then
    ...
else
    ...
endif

5. Perulangan (Looping)

// Perulangan FOR (Naik / Turun):
for <var> <- <awal> to <akhir> do ... endfor
for <var> <- <awal> downto <akhir> do ... endfor

// Perulangan WHILE:
while <kondisi> do ... endwhile

// Perulangan REPEAT - UNTIL:
repeat ... until <kondisi>

6. Sub-program (Prosedur & Fungsi)

// Prosedur (bisa dengan input/output / inout):
procedure <nama>(input/output <param> : <tipe>)
kamus
    // variabel lokal (opsional)
algoritma
    // instruksi prosedur
endprocedure

// Fungsi (dengan tipe pengembalian & return):
function <nama>(<param> : <tipe>) -> <tipe>
algoritma
    return <nilai>
endfunction

7. Padanan Sintaks Golang (Go)

// Input & Output Go:
fmt.Scan(&a, &b)
fmt.Println("Halo Dunia")

// Loop di Go:
for i := 1; i <= n; i++ { ... }
for <kondisi> { ... }

// Subprogram di Go:
func luas(r float64) float64 { return 3.14 * r * r }
func tukar(x, y *int) { ... }

8. Contoh Program Lengkap Siap Dijalankan

program hitung_lingkaran

    const PI = 3.14
    r, hasil : real

    function hitungLuas(radius : real) -> real
    algoritma
        return PI * radius * radius
    endfunction
