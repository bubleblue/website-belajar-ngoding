Kamus Sintaks Pseudocode & Golang

1. Struktur Dasar Pseudocode

program #####
kamus
    a, b : integer
    nama : string
algoritma
    // instruksi program ditulis di sini
endprogram

2. Struktur Dasar Golang (Go)

package main

import "fmt"

func main() {
    var a, b int
    var nama string
    // instruksi program di sini
}

3. I/O Pseudocode (input & output)

// menerima masukan:
input(nama)
input(a, b)

// menampilkan keluaran:
output("halo dunia")
output("nilai: ", a)

4. I/O Golang (fmt.Scan & fmt.Println)

// menerima masukan dari terminal:
fmt.Scan(&nama)
fmt.Scan(&a, &b)

// menampilkan keluaran ke terminal:
fmt.Println("halo dunia")
fmt.Printf("nilai a: %d\n", a)

5. Percabangan (Pseudocode vs Go)

// Pseudocode:
jika nilai >= 75 maka
    output("lulus")
lainnya
    output("remidi")
selesai_jika

// Golang:
if nilai >= 75 {
    fmt.Println("lulus")
} else {
    fmt.Println("remidi")
}

6. Perulangan Loop (Pseudocode vs Go)

// Pseudocode:
untuk i <- 1 sampai 5 lakukan
    output("angka: ", i)
selesai_untuk

// Golang:
for i := 1; i <= 5; i++ {
    fmt.Println("angka:", i)
}

7. While Loop (selama ... lakukan)

// Pseudocode:
selama i <= 5 lakukan
    output(i)
    i <- i + 1
selesai_selama

// Golang:
for i <= 5 {
    fmt.Println(i)
    i++
}

8. Contoh Program Golang Siap Dijalankan

package main

import "fmt"

func main() {
    var p, l int
    fmt.Println("=== Hitung Luas ===")
    fmt.Scan(&p)
    fmt.Scan(&l)
    luas := p * l
    fmt.Println("Luas Persegi Panjang:", luas)
}
